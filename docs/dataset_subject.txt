-- create the table

create table subjects_dataset_1 (primary key(gridimage_id))
select gi.gridimage_id,gi.moderation_status,GROUP_CONCAT(distinct tag limit 1) as subject, gi.imageclass,
IF(natnorthings>0 AND viewpoint_eastings>0,
pow(2,floor(log2(SQRT(
pow(cast(nateastings as signed)-cast(viewpoint_eastings as signed),2)
+pow(cast(natnorthings as signed)-cast(viewpoint_northings as signed),2)
))))
,'Unknown') AS distance,RAND() as rnd,RAND() as rnd2
from gridimage_search gi
inner join gridimage g using (gridimage_id)
left join tag_public t on (t.gridimage_id = gi.gridimage_id and t.prefix = 'subject' and canonical =0)
group by gridimage_id;

alter table subjects_dataset_1 add split enum('tra','val','tes') default null, add v tinyint unsigned default 0, add index(v);

update subjects_dataset_1 set split = 'tra' where rnd between 0 and 0.8;
update subjects_dataset_1 set split = 'val' where rnd between 0.8 and 0.9;
update subjects_dataset_1 set split = 'tes' where rnd between 0.9 and 1;


-- create the basic v0 dataset! (at the moment dont try to balance the classes!)
update subjects_dataset_1 set v=1 where v=0 order by rnd2 limit 1000;

-- backfill tentative subjects

alter table subjects_dataset_1 add imageclass varchar(64) default null after subject, add weight float not null default 1.0 after imageclass;

select gridimage_id,  gi.imageclass, m.subject, if(s.subject is null and m.subject != '', m.subject, s.subject), if (m.subject is not null and m.subject != '', 0.5, 1) as weight
 from subjects_dataset_1 s inner join gridimage_search gi using (gridimage_id) inner join category_mapping m on (m.imageclass = gi.imageclass) where gi.imageclass!='' limit 40;


update subjects_dataset_1 s inner join gridimage_search gi using (gridimage_id) inner join category_mapping m on (m.imageclass = gi.imageclass)
set s.imageclass = gi.imageclass -- always set
, s.subject = if(s.subject is null and m.subject != '', m.subject, s.subject) -- only set the subject if there isnt one, if no mapping, still sets NULL!
, weight = if(m.subject is not null and m.subject != '', 0.5, weight) -- if a mapped subject lower the weight! (note do this even if already subject tag as it likly came from mapping!)
where gi.imageclass != '';

## now, much later after having built the model, import the results back into the dataset table, mainly so can do easy comparsions
#(the dataset table is a convient record of what the subject was before model run, but also if the image was actully in training data) 
# ( because at some point contributor might start using the AI suggested tags to add them their historic images!)

alter table subjects_dataset_1 add ai_result varchar(1024) default null;


create table subjects_ai_result (primary key(gridimage_id))
select gridimage_id, max(score) as max_score
,coalesce(group_concat(if(score > 0.4,label,null) order by score desc separator '; '), group_concat(label order by score desc limit 1)) as ai_result
from gridimage_label where model = 'subjects' and gridimage_id < 1000 group by gridimage_id ;


update types_dataset_1 d inner join types_ai_result r using (gridimage_id) set d.ai_result = if(max_score < 0.5,concat(r.ai_result,' (unsure)'),r.ai_result);


create table subjects_ai_result (primary key(gridimage_id))
select gridimage_id, max(score) as max_score, group_concat(label order by score desc limit 1) as top1,
group_concat(concat(label,' ',round(score*100),'%') order by score desc separator '; ') as ai_result,
max(seq_id) as max_seq_id
from gridimage_label where model = 'subjects' group by gridimage_id ;

-- Can call many times to add new records - while processing is running
{
	select @max := max(max_seq_id) from subjects_ai_result;

	insert ignore into subjects_ai_result
	select gridimage_id, 0 as user_id, max(score) as max_score, group_concat(label order by score desc limit 1) as top1,
	group_concat(concat(label,' ',round(score*100),'%') order by score desc separator '; ') as ai_result,
	max(seq_id) as max_seq_id
	from gridimage_label where model = 'subjects' and seq_id > @max_id group by gridimage_id 

ON DUPLICATE KEY UPDATE 
    max_score = VALUES(max_score),
    top1 = VALUES(top1),
    ai_result = VALUES(ai_result),
    max_seq_id = VALUES(max_seq_id);

	update subjects_ai_result a inner join gridimage_search gi using (gridimage_id) set a.user_id = gi.user_id where a.user_id = 0;
}


alter table subjects_ai_result add user_id int unsigned not null after gridimage_id;
update subjects_ai_result a inner join gridimage_search gi using (gridimage_id) set a.user_id = gi.user_id;
alter table subjects_ai_result add index(user_id);


alter table subjects_ai_result add processed datetime default null;
