alter table places_dataset_1 add split enum('tra','val','tes') default null, add v tinyint unsigned default 0, add index(v);

update places_dataset_1 SET split = 'val' WHERE rnd1 > 0.8;

-- sample by hectad first
UPDATE places_dataset_1
JOIN (
    SELECT gridimage_id,
           ROW_NUMBER() OVER (PARTITION BY hectad ORDER BY rnd2) as rank_idx
    FROM places_dataset_1 WHERE v=0
) AS sampled ON places_dataset_1.gridimage_id = sampled.gridimage_id
SET v = 1
WHERE sampled.rank_idx <= 100;


-- then add images for under provisioned tags!
UPDATE places_dataset_1
JOIN (
    SELECT p.gridimage_id,
           ROW_NUMBER() OVER (PARTITION BY p.tag_id ORDER BY p.rnd2) as tag_rank,
           stats.already_selected
    FROM places_dataset_1 p
    JOIN (
        -- Calculate how many we already have for each tag
        SELECT tag_id, COUNT(*) as already_selected 
        FROM places_dataset_1 
        WHERE v = 1 
        GROUP BY tag_id
    ) AS stats ON p.tag_id = stats.tag_id
    WHERE p.v = 0
) AS topper ON places_dataset_1.gridimage_id = topper.gridimage_id
SET v = 1
WHERE topper.tag_rank <= (200 - topper.already_selected);
