CREATE FUNCTION get_largest_tier(
    p_max_size INT UNSIGNED
)
RETURNS INT UNSIGNED
DETERMINISTIC
BEGIN
    -- Apply the tiered logic using the single maximum value
    RETURN CASE
        -- Highest Tiers
        WHEN p_max_size >= 8192 THEN 8192
        WHEN p_max_size >= 3000 THEN 3000
        WHEN p_max_size >= 1600 THEN 1600
        WHEN p_max_size >= 1024 THEN 1024
        WHEN p_max_size >= 800  THEN 800
        
        -- Special case around 640, want to know if bigger, or explicitly just 640
        WHEN p_max_size > 640  THEN 641
        WHEN p_max_size = 640  THEN 640
        
        -- Lower Tiers 
        WHEN p_max_size >= 480 THEN 480
        WHEN p_max_size >= 320 THEN 320

        -- Final Fallback
        ELSE 96
    END;
END //

DELIMITER ;
